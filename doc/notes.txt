install docker on CentOS AlmaLinux
sudo dnf install -y yum-utils
sudo yum-config-manager --add-repo https://download.docker.com/linux/centos/docker-ce.repo
yum repolist | grep docker
sudo yum install -y docker-ce docker-ce-cli containerd.io
sudo systemctl enable docker
sudo systemctl start docker

# add current user to docker group
sudo usermod -aG docker $USER
# make it effective
newgrp docker

Edit /etc/yum.repos.d/CentOS-Base.repo
comment out mirrorlist for [base], [updates], [extras]
use baseurl instead
[base]
baseurl=http://vault.centos.org/7.9.2009/os/$basearch/
[updates]
baseurl=http://vault.centos.org/7.9.2009/updates/$basearch/
[extras]
baseurl=http://vault.centos.org/7.9.2009/extras/$basearch/

Some more is at: https://archive.fedoraproject.org/pub/archive/epel/testing/7/source/tree/

sudo yum clean all

sudo yum install epel-release -y \
    cmake llvm llvm-devel clang lld lldb \
    openssl openssl-devel boost-devel git

sudo vi /etc/systemd/logind.conf
HandleLidSwitch=ignore
HandleLidSwitchExternalPower=ignore

sudo systemctl restart systemd-logind

sudo hostnamectl set-hostname alma1

# for sanitizers
sudo dnf install gcc gcc-c++ libasan libubsan libtsan

Ubuntu setup:
sudo apt update
# libboost-all-dev has more
sudo apt install -y openssh-server net-tools git cmake llvm clang lldb lld openssl libssl-dev libboost-dev libasan6

# for claude:
sudo apt install -y nodejs npm
sudo npm install -g @anthropic-ai/claude-code

sudo systemctl enable --now ssh
sudo systemctl status ssh

# for laptop lid close
sudo vi /etc/systemd/logind.conf
HandleLidSwitch=ignore # do nothing when lid is closed
sudo systemctl restart systemd-logind

# error: attach failed: The current value of ptrace_scope is 1, which can cause ptrace to fail to attach to a running process. To fix this, run:
sudo sysctl -w kernel.yama.ptrace_scope=0

# install docker
sudo apt-get update
#optional
sudo apt-get install ca-certificates curl gnupg

#Add Docker’s official GPG key
sudo install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
sudo chmod a+r /etc/apt/keyrings/docker.gpg

#Add Docker repository
echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] \
  https://download.docker.com/linux/ubuntu \
  $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# update package list again
sudo apt-get update
sudo apt-get install docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

# make regular user to run docker
sudo usermod -aG docker $USER
# then logout and relogin
